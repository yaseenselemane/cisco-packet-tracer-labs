! R1 - office router (labs 2, 3 and 4)
hostname R1
no ip domain-lookup
!
! Lab 2: router-on-a-stick, one sub-interface per VLAN
interface g0/1
 description Trunk to SW1
 no shutdown
interface g0/1.10
 description SALES
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
interface g0/1.20
 description IT
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip nat inside
interface g0/1.30
 description GUEST
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip nat inside
 ip access-group GUEST-IN in
!
! Lab 3: DHCP for every VLAN
ip dhcp excluded-address 192.168.10.1 192.168.10.9
ip dhcp excluded-address 192.168.20.1 192.168.20.9
ip dhcp excluded-address 192.168.20.100
ip dhcp excluded-address 192.168.30.1 192.168.30.9
ip dhcp pool SALES
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
ip dhcp pool IT
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
ip dhcp pool GUEST
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
!
! Lab 3: NAT overload (PAT) out to the ISP
interface g0/0
 description Link to ISP
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no shutdown
access-list 1 permit 192.168.0.0 0.0.255.255
ip nat inside source list 1 interface g0/0 overload
ip route 0.0.0.0 0.0.0.0 203.0.113.1
!
! Lab 4: guests get the internet, not the office
ip access-list extended GUEST-IN
 remark Guests may use the internet but not the office VLANs
 deny ip 192.168.30.0 0.0.0.255 192.168.10.0 0.0.0.255
 deny ip 192.168.30.0 0.0.0.255 192.168.20.0 0.0.0.255
 permit ip any any
